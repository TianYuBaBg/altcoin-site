"""Telegram notification sender via HTTP API (zero extra dependencies)."""
import requests

from opportunity.config import TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID


def send(text: str) -> bool:
    """Send a plain text message to the configured Telegram chat.

    Returns True on success, False on failure (logs error to stderr).
    """
    if not TELEGRAM_BOT_TOKEN or not TELEGRAM_CHAT_ID:
        print("[notifier] TELEGRAM_BOT_TOKEN or TELEGRAM_CHAT_ID not configured")
        return False

    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    try:
        r = requests.post(url, json={
            "chat_id": TELEGRAM_CHAT_ID,
            "text": text,
        }, timeout=10)
        if not r.ok:
            print(f"[notifier] Telegram API error: {r.status_code} {r.text[:200]}")
        return r.ok
    except requests.RequestException as e:
        print(f"[notifier] Request failed: {e}")
        return False

"""Dedup state management for opportunity monitor.

Persists to data/opportunity/state.json so notifications don't repeat.
State shape:
{
  "notified_tids": [int, ...],          # new project signals already sent
  "prices": {"TICKER": {"last": float, "notified_at": str}},
  "listings": {"known_ids": [str, ...]},
  "community": {"TID": {"replies": int, "checked": str}},
  "github": {"user/repo": {"pushed_at": str, "notified": bool}}
}
"""
import json
from pathlib import Path

# Inline path — no need to modify paths.py for one constant
_STATE_FILE = Path(__file__).resolve().parent.parent.parent / 'data' / 'opportunity' / 'state.json'


def load() -> dict:
    """Load state from disk. Returns default empty state if file missing."""
    if _STATE_FILE.exists():
        return json.loads(_STATE_FILE.read_text(encoding='utf-8'))
    return {
        "notified_tids": [],
        "prices": {},
        "listings": {"known_ids": []},
        "community": {},
        "github": {},
    }


def save(state: dict):
    """Save state to disk, creating parent dirs as needed."""
    _STATE_FILE.parent.mkdir(parents=True, exist_ok=True)
    _STATE_FILE.write_text(json.dumps(state, ensure_ascii=False, indent=2), encoding='utf-8')

#!/usr/bin/env python
"""Fetch exchange prices for all tracked tickers.

Collects tickers from project .md filenames, queries MEXC public API,
and writes results to data/opportunity/prices.json.

Usage:
    python scripts/opportunity/fetch_exchange.py
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import requests
from utils.paths import DATA_DIR, PROJECTS_DIR

OUTPUT = DATA_DIR / 'opportunity' / 'prices.json'


def _collect_tickers() -> list[str]:
    """Extract ticker symbols from project .md filenames (TID_TICKER_NAME.md)."""
    tickers = set()
    for f in PROJECTS_DIR.glob('*.md'):
        if f.name == '总览.md':
            continue
        parts = f.stem.split('_')
        if len(parts) >= 2 and parts[0].isdigit() and parts[1].isalnum():
            tickers.add(parts[1].upper())
    return sorted(tickers)


def _query_mexc(ticker: str) -> dict | None:
    """Query MEXC public API for ticker price on USDT and BTC pairs."""
    for quote in ['USDT', 'BTC']:
        try:
            url = f"https://api.mexc.com/api/v3/ticker/price?symbol={ticker}{quote}"
            r = requests.get(url, timeout=10)
            if r.ok:
                data = r.json()
                if 'price' in data:
                    return {'price': float(data['price']), 'exchange': 'MEXC', 'pair': f'{ticker}/{quote}'}
        except Exception:
            pass
    return None


def main():
    tickers = _collect_tickers()
    print(f"[fetch_exchange] Tracking {len(tickers)} tickers")

    prices = {}
    for ticker in tickers:
        result = _query_mexc(ticker)
        if result:
            prices[ticker] = result

    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(json.dumps(prices, ensure_ascii=False, indent=2), encoding='utf-8')
    print(f"[fetch_exchange] Updated {len(prices)} prices → {OUTPUT}")


if __name__ == '__main__':
    main()

#!/usr/bin/env python
"""Fetch GitHub repo metadata for all repos referenced in project .md files.

Uses existing fetch_github_api from utils.fetcher (which respects GITHUB_TOKEN env var).
Outputs to data/opportunity/github_data.json.

Usage:
    python scripts/opportunity/fetch_github.py
"""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from utils.fetcher import fetch_github_api
from utils.paths import DATA_DIR, PROJECTS_DIR

OUTPUT = DATA_DIR / 'opportunity' / 'github_data.json'


def _collect_repos() -> set[str]:
    """Extract unique GitHub repos from project .md files."""
    repos = set()
    for f in PROJECTS_DIR.glob('*.md'):
        if f.name == '总览.md':
            continue
        content = f.read_text(encoding='utf-8')
        for m in re.finditer(r'github\.com/([^/\s"<>]+/[^/\s"<>#?]+)', content):
            repos.add(m.group(1).rstrip('/').lower())
    return repos


def main():
    repos = _collect_repos()
    print(f"[fetch_github] Found {len(repos)} unique repos")

    result = {}
    for repo in sorted(repos):
        data = fetch_github_api(repo)
        if data and not data.get('message', '').startswith('Not Found'):
            result[repo] = {
                'stars': data.get('stargazers_count', 0),
                'forks': data.get('forks_count', 0),
                'pushed_at': data.get('pushed_at', ''),
                'language': data.get('language') or '',
                'description': (data.get('description') or '')[:120],
            }

    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding='utf-8')
    print(f"[fetch_github] Updated {len(result)} repos → {OUTPUT}")


if __name__ == '__main__':
    main()

#!/usr/bin/env python
"""Fetch current exchange listings and cross-reference with tracked tickers.

Queries MEXC exchangeInfo to find which tracked tickers are listed.
Outputs to data/opportunity/listings.json.

Usage:
    python scripts/opportunity/fetch_listings.py
"""
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import requests
from utils.paths import DATA_DIR, PROJECTS_DIR

OUTPUT = DATA_DIR / 'opportunity' / 'listings.json'


def _collect_tickers() -> set[str]:
    """Extract tickers from project .md filenames."""
    tickers = set()
    for f in PROJECTS_DIR.glob('*.md'):
        if f.name == '总览.md':
            continue
        parts = f.stem.split('_')
        if len(parts) >= 2 and parts[0].isdigit() and parts[1].isalnum():
            tickers.add(parts[1].upper())
    return tickers


def _fetch_mexc_listings(tracked: set[str]) -> list[dict]:
    """Fetch all MEXC trading pairs and match against tracked tickers."""
    found = []
    try:
        r = requests.get("https://api.mexc.com/api/v3/exchangeInfo", timeout=15)
        if r.ok:
            symbols = r.json().get('symbols', [])
            for s in symbols:
                base = s.get('baseAsset', '')
                quote = s.get('quoteAsset', '')
                if base in tracked:
                    found.append({
                        'base': base,
                        'exchange': 'MEXC',
                        'pair': f'{base}/{quote}',
                        'status': s.get('status', ''),
                    })
    except requests.RequestException as e:
        print(f"[fetch_listings] MEXC error: {e}")
    return found


def main():
    tickers = _collect_tickers()
    print(f"[fetch_listings] Tracking {len(tickers)} tickers")

    listings = _fetch_mexc_listings(tickers)
    print(f"[fetch_listings] Found {len(listings)} matching listings")

    OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT.write_text(json.dumps(listings, ensure_ascii=False, indent=2), encoding='utf-8')
    print(f"[fetch_listings] Written to {OUTPUT}")


if __name__ == '__main__':
    main()

"""5 signal detectors for Module 5 opportunity monitor.

Each check_* function reads data files, compares with state dict,
appends notification text to a list, and mutates state for dedup.
All functions return list[str] (0 or more notification messages).
"""
import json
import re
from datetime import datetime
from pathlib import Path

from opportunity.config import MIN_SCORE, PRICE_CHANGE_PERCENT, REPLY_INCREASE
from utils.paths import DATA_DIR, PROJECTS_DIR

LOG_DIR = DATA_DIR / 'pipeline_logs'
PRICES_FILE = DATA_DIR / 'opportunity' / 'prices.json'
LISTINGS_FILE = DATA_DIR / 'opportunity' / 'listings.json'
GITHUB_FILE = DATA_DIR / 'opportunity' / 'github_data.json'

# ── helpers ─────────────────────────────────────────────────

_TICKER_CN_CACHE: dict[str, str] | None = None


def _build_ticker_cache():
    global _TICKER_CN_CACHE
    if _TICKER_CN_CACHE is not None:
        return
    _TICKER_CN_CACHE = {}
    for f in PROJECTS_DIR.glob('*.md'):
        if f.name == '总览.md':
            continue
        parts = f.stem.split('_')
        if len(parts) >= 3 and parts[0].isdigit():
            _TICKER_CN_CACHE[parts[1].upper()] = parts[2]


def _cn_name(ticker: str) -> str:
    _build_ticker_cache()
    return _TICKER_CN_CACHE.get(ticker.upper(), '')


def _find_md_by_tid(tid: int) -> Path | None:
    for f in PROJECTS_DIR.glob(f'{tid}_*.md'):
        return f
    for f in PROJECTS_DIR.glob(f'_{tid}.md'):
        return f
    return None


def _quick_score(replies: int, views: int, last_active: str, exchange: str) -> int:
    """Simplified 0-12 score based on available .md metadata fields."""
    score = 0
    if replies >= 200: score += 3
    elif replies >= 50: score += 2
    elif replies >= 10: score += 1

    if views >= 50000: score += 3
    elif views >= 10000: score += 2
    elif views >= 1000: score += 1

    if last_active:
        try:
            days = (datetime.now() - datetime.strptime(last_active, '%Y-%m-%d')).days
            if days <= 30: score += 3
            elif days <= 90: score += 2
            elif days <= 365: score += 1
        except ValueError:
            pass

    if exchange and exchange != '未上架':
        score += 3

    return score


def _parse_md(filepath: Path) -> dict:
    """Parse key fields from a project .md file."""
    content = filepath.read_text(encoding='utf-8')
    result = {'replies': 0, 'views': 0, 'status': '', 'last_active': '', 'exchange': ''}

    m = re.search(r'\*\*回复数[:**：]*\s*([\d,]+)', content)
    if m: result['replies'] = int(m.group(1).replace(',', ''))

    m = re.search(r'\*\*浏览量[:**：]*\s*([\d,]+)', content)
    if m: result['views'] = int(m.group(1).replace(',', ''))

    m = re.search(r'\*\*状态[:**：]*\s*([^\n|]+)', content)
    if m: result['status'] = m.group(1).strip()

    m = re.search(r'\*\*最后活动[:**：]*\s*(\d{4}-\d{2}-\d{2})', content)
    if m: result['last_active'] = m.group(1)

    m = re.search(r'\*\*交易所[:**：]*\s*([^\n|]+)', content)
    if m: result['exchange'] = m.group(1).strip()

    result['title'] = content.split('\n')[0].replace('# ', '').strip() if content else ''
    return result


# ── signal detectors ─────────────────────────────────────────

def check_new_projects(state: dict) -> list[str]:
    """Detect projects that recently completed module 3/4 with score >= MIN_SCORE."""
    notifications = []
    notified = set(state.get('notified_tids', []))
    now = datetime.now().isoformat()

    for log_file in sorted(LOG_DIR.glob('*.json'), reverse=True):
        tid = int(log_file.stem)
        if tid in notified:
            continue

        log = json.loads(log_file.read_text(encoding='utf-8'))
        modules = log.get('模块', {})
        m3 = modules.get('模块3', {}).get('状态', '')
        m4 = modules.get('模块4', {}).get('状态', '')
        if m3 != '已完成' and m4 != '已完成':
            continue

        md_file = _find_md_by_tid(tid)
        if not md_file:
            continue

        meta = _parse_md(md_file)
        score = _quick_score(meta['replies'], meta['views'], meta['last_active'], meta['exchange'])
        if score < MIN_SCORE:
            continue

        parts = md_file.stem.split('_')
        ticker = parts[1] if len(parts) >= 2 and parts[0].isdigit() else ''
        cn_name = parts[2] if len(parts) >= 3 else parts[0]
        title = log.get('标题', meta['title'])[:80]

        msg = (
            f"🆕 新项目: {cn_name} ({ticker})\n"
            f"━━━━━━━━━━━━━━━━━━\n"
            f"📊 评分: {score}/12\n"
            f"📌 状态: {meta['status']}\n"
            f"💬 回复: {meta['replies']} | 👁 浏览: {meta['views']:,}\n"
            f"🏷 {title}\n"
            f"🔗 bitcointalk.org/index.php?topic={tid}.0"
        )
        notifications.append(msg)
        notified.add(tid)

    state['notified_tids'] = sorted(notified)
    return notifications


def check_price_moves(state: dict) -> list[str]:
    """Detect price changes >= PRICE_CHANGE_PERCENT compared to last check."""
    notifications = []
    if not PRICES_FILE.exists():
        return notifications

    prices = json.loads(PRICES_FILE.read_text(encoding='utf-8'))
    price_state = state.get('prices', {})

    for ticker, data in prices.items():
        current = data['price']
        prev_entry = price_state.get(ticker, {})
        prev = prev_entry.get('last')

        if prev is not None and prev > 0:
            change = (current - prev) / prev * 100
            if abs(change) >= PRICE_CHANGE_PERCENT:
                emoji = '📈' if change > 0 else '📉'
                sign = '+' if change > 0 else ''
                direction = '上涨' if change > 0 else '下跌'
                cn = _cn_name(ticker)

                msg = (
                    f"{emoji} 价格{direction}: {ticker} {sign}{change:.1f}%\n"
                    f"━━━━━━━━━━━━━━━━━━\n"
                    f"当前: {current:.8f}\n"
                    f"上次: {prev:.8f}\n"
                    f"🔄 {data.get('exchange', '未知')} | {data.get('pair', '')}"
                )
                notifications.append(msg)

        price_state[ticker] = {'last': current, 'notified_at': datetime.now().isoformat()}

    state['prices'] = price_state
    return notifications


def check_new_listings(state: dict) -> list[str]:
    """Detect exchange listings not previously notified."""
    notifications = []
    if not LISTINGS_FILE.exists():
        return notifications

    listings = json.loads(LISTINGS_FILE.read_text(encoding='utf-8'))
    known_ids = set(state.get('listings', {}).get('known_ids', []))
    current_ids = set()

    for listing in listings:
        lid = f"{listing['base']}_{listing['exchange']}_{listing.get('pair', '')}"
        current_ids.add(lid)

        if lid not in known_ids:
            cn = _cn_name(listing['base'])
            msg = (
                f"📋 新上交易所\n"
                f"━━━━━━━━━━━━━━━━━━\n"
                f"币种: {listing['base']} ({cn})\n"
                f"平台: {listing['exchange']}\n"
                f"交易对: {listing.get('pair', listing['base'] + '/USDT')}"
            )
            notifications.append(msg)

    state['listings'] = {'known_ids': sorted(current_ids)}
    return notifications


def check_community_activity(state: dict) -> list[str]:
    """Detect reply count increases >= REPLY_INCREASE since last check."""
    notifications = []
    community_state = state.get('community', {})
    now = datetime.now().isoformat()

    for md_file in sorted(PROJECTS_DIR.glob('*.md')):
        if md_file.name == '总览.md':
            continue

        stem = md_file.stem
        tid_str = stem.split('_')[0]
        if not tid_str.isdigit():
            continue
        tid = int(tid_str)

        meta = _parse_md(md_file)
        current = meta['replies']
        prev_entry = community_state.get(tid, {})
        prev = prev_entry.get('replies', 0)

        if prev > 0:
            increase = current - prev
            if increase >= REPLY_INCREASE:
                parts = stem.split('_')
                ticker = parts[1] if len(parts) >= 2 else ''
                cn_name = parts[2] if len(parts) >= 3 else parts[0]

                msg = (
                    f"💬 社区活跃: {cn_name} ({ticker})\n"
                    f"━━━━━━━━━━━━━━━━━━\n"
                    f"📈 新增 +{increase} 回复\n"
                    f"📊 总计: {current} 回复 / {meta['views']:,} 浏览\n"
                    f"📅 最后活动: {meta['last_active']}\n"
                    f"🔗 bitcointalk.org/index.php?topic={tid}.0"
                )
                notifications.append(msg)

        community_state[tid] = {'replies': current, 'checked': now}

    state['community'] = community_state
    return notifications


def check_github_activity(state: dict) -> list[str]:
    """Detect GitHub repos with newer pushed_at since last check."""
    notifications = []
    if not GITHUB_FILE.exists():
        return notifications

    gh_data = json.loads(GITHUB_FILE.read_text(encoding='utf-8'))
    gh_state = state.get('github', {})

    for repo, data in gh_data.items():
        current_pushed = data.get('pushed_at', '')
        if not current_pushed:
            continue

        prev = gh_state.get(repo, {})
        prev_pushed = prev.get('pushed_at', '')

        if current_pushed != prev_pushed:
            ticker = ''
            for f in PROJECTS_DIR.glob('*.md'):
                if f.name == '总览.md':
                    continue
                if repo.lower() in f.read_text(encoding='utf-8').lower():
                    parts = f.stem.split('_')
                    if len(parts) >= 2 and parts[0].isdigit():
                        ticker = parts[1]
                        break

            msg = (
                f"🔧 GitHub 更新: {ticker or repo}\n"
                f"━━━━━━━━━━━━━━━━━━\n"
                f"📦 {repo}\n"
                f"⭐ {data.get('stars', 0)} stars | 🍴 {data.get('forks', 0)} forks\n"
                f"📝 语言: {data.get('language', '未知')}\n"
                f"🔗 https://github.com/{repo}"
            )
            notifications.append(msg)

        gh_state[repo] = {'pushed_at': current_pushed}

    state['github'] = gh_state
    return notifications

python
#!/usr/bin/env python
"""Module 5: Opportunity Monitor — main entry point.

Orchestrates data fetching, signal detection, and Telegram notifications
in a continuous loop. Can also run once for manual/cron invocation.

Usage:
    python scripts/opportunity/monitor.py              # continuous loop
    python scripts/opportunity/monitor.py --once       # single run
    python scripts/opportunity/monitor.py --interval 5 # custom interval
"""
import argparse
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from opportunity.config import CHECK_INTERVAL_MINUTES
from opportunity.state import load, save
from opportunity.notifier import send
from opportunity.signals import (
    check_new_projects,
    check_price_moves,
    check_new_listings,
    check_community_activity,
    check_github_activity,
)
from opportunity import fetch_exchange, fetch_github, fetch_listings


def run_once():
    """Execute one full check cycle: fetch → detect → notify → save."""
    print("[monitor] === Opportunity check cycle ===")

    # 1. Fetch fresh data
    print("[monitor] Fetching exchange prices...")
    fetch_exchange.main()
    print("[monitor] Fetching GitHub data...")
    fetch_github.main()
    print("[monitor] Fetching exchange listings...")
    fetch_listings.main()

    # 2. Load state & detect signals
    state = load()

    all_signals: list[str] = []
    all_signals.extend(check_new_projects(state))
    all_signals.extend(check_price_moves(state))
    all_signals.extend(check_new_listings(state))
    all_signals.extend(check_community_activity(state))
    all_signals.extend(check_github_activity(state))

    # 3. Send notifications (with 1s delay between to avoid rate limits)
    sent = 0
    for msg in all_signals:
        print(f"[monitor] Sending signal ({sent + 1}/{len(all_signals)})")
        if send(msg):
            sent += 1
        time.sleep(1)

    # 4. Save updated state
    save(state)
    print(f"[monitor] Cycle complete: {sent} signal(s) sent")
    return sent


def main():
    parser = argparse.ArgumentParser(description="Module 5: Opportunity Monitor")
    parser.add_argument('--once', action='store_true', help='Run once then exit')
    parser.add_argument('--interval', type=int, default=CHECK_INTERVAL_MINUTES,
                        help='Check interval in minutes')
    args = parser.parse_args()

    if args.once:
        run_once()
        return

    interval = max(1, args.interval)
    print(f"[monitor] Starting continuous loop (interval={interval}min)")
    while True:
        run_once()
        print(f"[monitor] Sleeping {interval} minutes...")
        time.sleep(interval * 60)


if __name__ == '__main__':
    main()
